Build the agent-first listing workflow with investor/off-market deal support. This is one comprehensive build — read everything before starting.

The Core Use Case

A realtor finds a fixer or non-loanable property. They create a private listing on xucasa, get a shareable link, and send it to investor contacts via text/DM/email. The investor taps the link, lands on xucasa.com, sees the listing with the agent's contact info front and center, and can call/text/message the agent with zero friction — no account required. If no deal happens off-market, the property goes to MLS, the off-market listing auto-archives, and the MLS listing takes over on the portal. The agent sees everything in their dashboard with analytics.

Auto-archive in IDX sync (server/idxSync.ts or artifacts/api-server/src/idxSync.ts)

In the IDX sync upsert logic, after each property is upserted from the MLS feed:

typescript
// After upserting an MLS listing, check for matching off-market/coming-soon listings
async function checkAndArchiveOffMarketMatch(mlsProperty: Property) {
  // Match by: expectedMlsNumber (exact) OR address (street number + name + city, normalized)
  const normalizeAddress = (s: string) => s.toLowerCase().trim().replace(/\s+/g, ' ');

  const candidates = await db.select()
    .from(properties)
    .where(
      and(
        inArray(properties.listingStatus, ['off_market', 'coming_soon', 'pre_market']),
        or(
          // MLS number match (most reliable)
          eq(properties.expectedMlsNumber, mlsProperty.mlsNumber),
          // Address match (fallback)
          and(
            eq(
              sql`LOWER(TRIM(${properties.addressStreetNumber} || ' ' || ${properties.addressStreetName}))`,
              normalizeAddress(`${mlsProperty.addressStreetNumber} ${mlsProperty.addressStreetName}`)
            ),
            eq(sql`LOWER(TRIM(${properties.addressCity}))`, normalizeAddress(mlsProperty.addressCity))
          )
        )
      )
    );

  for (const offMarketListing of candidates) {
    // Archive the off-market listing
    await db.update(properties)
      .set({
        listingStatus: 'archived',
        archivedAt: new Date(),
        archivedReason: 'went_to_mls',
        linkedMlsPropertyId: mlsProperty.id,
      })
      .where(eq(properties.id, offMarketListing.id));

    // Notify the agent
    if (offMarketListing.agentId) {
      await storage.createNotification({
        userId: offMarketListing.agentId,
        type: 'listing_went_live',
        title: 'Your listing went live on MLS',
        message: `Your off-market listing at ${offMarketListing.addressStreetNumber} ${offMarketListing.addressStreetName} has gone live on the MLS and has been archived. The active MLS listing is now live on Xucasa.`,
        propertyId: mlsProperty.id,
        linkUrl: `/property/${mlsProperty.id}`,
        metadata: {
          archivedPropertyId: offMarketListing.id,
          mlsPropertyId: mlsProperty.id,
        },
      });
    }
  }
}

Call this function for every property upserted during IDX sync.

// Shows on mobile browsers only (not in-app, not on desktop)
// Appears after 8 seconds on the page or after scrolling 300px — whichever comes first
// Dismissable with X — stays dismissed for the session (React state only, no localStorage)
// Position: sticky bottom banner (above any other fixed elements)
// Does NOT show if user is already using the Xucasa iOS app (check user agent for Expo/xucasa)

const AppDownloadBanner = () => {
  const [show, setShow] = useState(false);
  const [dismissed, setDismissed] = useState(false);

  useEffect(() => {
    // Only show on mobile browsers
    const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent);
    const isInApp = /xucasa|Expo/i.test(navigator.userAgent);
    if (!isMobile || isInApp || dismissed) return;

    // Show after 8 seconds
    const timer = setTimeout(() => setShow(true), 8000);

    // Or after 300px scroll
    const handleScroll = () => {
      if (window.scrollY > 300) { setShow(true); clearTimeout(timer); }
    };
    window.addEventListener('scroll', handleScroll);
    return () => { clearTimeout(timer); window.removeEventListener('scroll', handleScroll); };
  }, [dismissed]);

  if (!show || dismissed) return null;

  return (
    <div className="fixed bottom-0 left-0 right-0 z-50 bg-white dark:bg-card border-t shadow-lg px-4 py-3 flex items-center gap-3">
      <img src="/icons/icon-72x72.png" className="w-12 h-12 rounded-xl shrink-0" />
      <div className="flex-1 min-w-0">
        <div className="font-semibold text-sm">Xucasa</div>
        <div className="text-xs text-muted-foreground">Better on the app — search, swipe, connect</div>
      </div>
      
        href="https://apps.apple.com/app/xucasa/PLACEHOLDER"
        className="bg-amber-500 text-white text-xs font-medium px-3 py-1.5 rounded-full shrink-0"
      >
        Get App
      </a>
      <button onClick={() => { setDismissed(true); setShow(false); }} className="text-muted-foreground shrink-0 ml-1">
        <X className="w-4 h-4" />
      </button>
    </div>
  );
};
