typescriptrouter.post('/onboarding/lender', onboardingRateLimit, isAuthenticated, async (req: any, res) => {
  const schema = z.object({
    companyName: z.string().min(1),
    nmls: z.string().optional(),
    licenseState: z.string().optional(),
    specialties: z.array(z.string()).optional().default([]),
    bio: z.string().optional(),
    phone: z.string().optional(),
    website: z.string().url().optional().or(z.literal('')),
  });
  const parsed = schema.safeParse(req.body);
  if (!parsed.success) {
    return res.status(400).json({ message: 'Invalid request', errors: parsed.error.flatten() });
  }

  const userId = req.user.claims.sub;

  await executeWithAudit({
    eventType: 'lender_onboarding_completed',
    actorUserId: userId,
    actorRole: 'lender',
    resourceType: 'lender_profile',
    outcome: 'success',
  }, async () => {
    await storage.upsertLenderProfile(userId, parsed.data);
    await storage.updateUser(userId, {
      onboardingCompleted: true,
      primaryIntent: 'lender',
    });
  });

  res.json({ destination: '/dashboard' });
});
