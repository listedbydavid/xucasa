Build an admin "Preview As User" mode toggle that lets David test the platform as a buyer, homeowner, agent, or explorer — without losing admin access or getting stuck in that mode.

How it should work

A floating toggle persists on screen for admin users only — always visible, never hidden by page content
Admin can select a role to preview: Buyer, Homeowner, Agent, Explorer
While previewing, the app renders exactly as that role would see it — nav, dashboard, onboarding routing, feature visibility all match that role
A prominent banner stays visible at ALL times during preview so David can never forget he's in preview mode
One click exits preview and returns to full admin mode
Preview state is stored in memory only (React state + sessionStorage) — never written to the database, never changes David's actual user record
If the page is refreshed during preview, preview mode is cleared and David returns to admin mode

tsximport { useState, useEffect } from 'react';
import { useLocation } from 'wouter';
import { Eye, X, ChevronDown } from 'lucide-react';
import { Button } from '@/components/ui/button';
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuTrigger,
} from '@/components/ui/dropdown-menu';

export type PreviewRole = 'buyer' | 'homeowner' | 'agent' | 'explorer' | null;

interface AdminPreviewBarProps {
  isAdmin: boolean;
  previewRole: PreviewRole;
  onPreviewChange: (role: PreviewRole) => void;
}

const ROLE_LABELS: Record<NonNullable<PreviewRole>, string> = {
  buyer: 'Buyer',
  homeowner: 'Homeowner',
  agent: 'Agent',
  explorer: 'Explorer',
};

const ROLE_DESTINATIONS: Record<NonNullable<PreviewRole>, string> = {
  buyer: '/swipe',
  homeowner: '/home-report',
  agent: '/agent',
  explorer: '/swipe',
};

const ROLE_COLORS: Record<NonNullable<PreviewRole>, string> = {
  buyer: 'bg-blue-500',
  homeowner: 'bg-green-500',
  agent: 'bg-purple-500',
  explorer: 'bg-gray-500',
};

export function AdminPreviewBar({ isAdmin, previewRole, onPreviewChange }: AdminPreviewBarProps) {
  const [, navigate] = useLocation();

  if (!isAdmin) return null;

  const handleActivate = (role: NonNullable<PreviewRole>) => {
    onPreviewChange(role);
    sessionStorage.setItem('xucasa_preview_role', role);
    navigate(ROLE_DESTINATIONS[role]);
  };

  const handleExit = () => {
    onPreviewChange(null);
    sessionStorage.removeItem('xucasa_preview_role');
    navigate('/admin');
  };

  // Floating launcher when not in preview
  if (!previewRole) {
    return (
      <div className="fixed bottom-20 right-4 z-50">
        <DropdownMenu>
          <DropdownMenuTrigger asChild>
            <Button
              size="sm"
              className="bg-slate-800 hover:bg-slate-700 text-white shadow-lg gap-2 rounded-full px-4"
            >
              <Eye className="w-4 h-4" />
              Preview as...
              <ChevronDown className="w-3 h-3" />
            </Button>
          </DropdownMenuTrigger>
          <DropdownMenuContent align="end" className="w-44">
            {(Object.keys(ROLE_LABELS) as NonNullable<PreviewRole>[]).map(role => (
              <DropdownMenuItem
                key={role}
                onClick={() => handleActivate(role)}
                className="gap-2 cursor-pointer"
              >
                <span className={`w-2 h-2 rounded-full ${ROLE_COLORS[role]}`} />
                {ROLE_LABELS[role]}
              </DropdownMenuItem>
            ))}
          </DropdownMenuContent>
        </DropdownMenu>
      </div>
    );
  }

  // Active preview banner — always visible, impossible to miss
  return (
    <>
      {/* Top banner */}
      <div className={`fixed top-0 left-0 right-0 z-[100] ${ROLE_COLORS[previewRole]} text-white`}>
        <div className="flex items-center justify-between px-4 py-2 max-w-7xl mx-auto">
          <div className="flex items-center gap-2 text-sm font-medium">
            <Eye className="w-4 h-4" />
            <span>Admin Preview — viewing as <strong>{ROLE_LABELS[previewRole]}</strong></span>
            <span className="opacity-75 text-xs">(your account is unchanged)</span>
          </div>
          <Button
            size="sm"
            variant="ghost"
            onClick={handleExit}
            className="text-white hover:bg-white/20 gap-1 h-7 text-xs"
          >
            <X className="w-3 h-3" />
            Exit Preview → Back to Admin
          </Button>
        </div>
      </div>
      {/* Spacer so page content doesn't hide under the banner */}
      <div className="h-10" />
      {/* Floating exit button — secondary escape hatch, always visible */}
      <div className="fixed bottom-20 right-4 z-50">
        <Button
          size="sm"
          onClick={handleExit}
          className="bg-slate-800 hover:bg-slate-700 text-white shadow-lg gap-2 rounded-full px-4"
        >
          <X className="w-4 h-4" />
          Exit Preview
        </Button>
      </div>
    </>
  );
}

Step B — Create preview context (client/src/lib/preview-context.tsx)
typescriptimport { createContext, useContext, useState, useEffect, ReactNode } from 'react';
import type { PreviewRole } from '@/components/AdminPreviewBar';

interface PreviewContextValue {
  previewRole: PreviewRole;
  setPreviewRole: (role: PreviewRole) => void;
  effectiveRole: string | null; // the role to use for UI decisions
  isPreviewActive: boolean;
}

const PreviewContext = createContext<PreviewContextValue>({
  previewRole: null,
  setPreviewRole: () => {},
  effectiveRole: null,
  isPreviewActive: false,
});

export function PreviewProvider({ children }: { children: ReactNode }) {
  // Restore from sessionStorage on mount (survives hot reload, cleared on tab close)
  const [previewRole, setPreviewRoleState] = useState<PreviewRole>(() => {
    try {
      const stored = sessionStorage.getItem('xucasa_preview_role');
      return (stored as PreviewRole) ?? null;
    } catch {
      return null;
    }
  });

  const setPreviewRole = (role: PreviewRole) => {
    setPreviewRoleState(role);
    if (role) {
      sessionStorage.setItem('xucasa_preview_role', role);
    } else {
      sessionStorage.removeItem('xucasa_preview_role');
    }
  };

  return (
    <PreviewContext.Provider value={{
      previewRole,
      setPreviewRole,
      effectiveRole: previewRole,
      isPreviewActive: previewRole !== null,
    }}>
      {children}
    </PreviewContext.Provider>
  );
}

export const usePreview = () => useContext(PreviewContext);
